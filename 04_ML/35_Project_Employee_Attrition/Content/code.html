In [25]cell 42
iqr_bounds = {}
for c in outlier_cols:
    q1, q3 = dataset[c].quantile(0.25), dataset[c].quantile(0.75)
    iqr = q3 - q1
    lo, hi = q1 - 1.5 * iqr, q3 + 1.5 * iqr
    iqr_bounds[c] = (lo, hi)
    n_out = ((dataset[c] < lo) | (dataset[c] > hi)).sum()
    print(f"{c}: IQR bounds=({lo:.1f}, {hi:.1f})  outliers={n_out}")
